    \node[font=\scriptsize, anchor=west, inner sep=0pt, text=black!70] at (-29,4.8) {Rows one unit blocks};
    \pic at (-29,0.2) {deckslots};
    \foreach \i in {1,...,#3} \fill[#2] (-29+\i*1.45-1.45,0.2) rectangle ++(1.2,3);

    \fill[black!18] (-5,0.5) rectangle (26,2.9);
    \fill[#2] (-5,0.5) rectangle ++(#4,2.4);
    \node[font=\scriptsize\bfseries, anchor=west, inner sep=1pt, text=#2] at (-5+#4,1.7) {#5};
  \end{scope}}
